cmake_minimum_required(VERSION 3.24)
project(switch_drive_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_path(ZSTD_INCLUDE_DIR zstd.h REQUIRED)
find_library(ZSTD_LIBRARY NAMES zstd REQUIRED)
find_path(MBEDTLS_INCLUDE_DIR mbedtls/aes.h REQUIRED)
find_library(MBEDCRYPTO_LIBRARY NAMES mbedcrypto REQUIRED)
find_package(CURL REQUIRED)
find_package(Threads REQUIRED)
find_path(JANSSON_INCLUDE_DIR jansson.h REQUIRED)
find_library(JANSSON_LIBRARY NAMES jansson REQUIRED)
add_executable(switch_drive_tests core_tests.cpp ../switch/source/app_controller.cpp ../switch/source/core.cpp ../switch/source/installer.cpp ../switch/source/nsz.cpp ../switch/source/download.cpp ../switch/source/network.cpp ../switch/source/i18n.cpp ../switch/source/ui_model.cpp ../switch/source/qr.cpp)
target_include_directories(switch_drive_tests PRIVATE ../switch/include ${ZSTD_INCLUDE_DIR} ${MBEDTLS_INCLUDE_DIR} ${JANSSON_INCLUDE_DIR})
target_link_libraries(switch_drive_tests PRIVATE ${ZSTD_LIBRARY} ${MBEDCRYPTO_LIBRARY} CURL::libcurl ${JANSSON_LIBRARY} Threads::Threads)
enable_testing()
add_test(NAME switch_drive_tests COMMAND switch_drive_tests)

add_executable(switch_drive_download_tests download_tests.cpp ../switch/source/core.cpp ../switch/source/download.cpp ../switch/source/network.cpp ../switch/source/i18n.cpp)
target_include_directories(switch_drive_download_tests PRIVATE ../switch/include ${JANSSON_INCLUDE_DIR})
target_link_libraries(switch_drive_download_tests PRIVATE CURL::libcurl ${JANSSON_LIBRARY} Threads::Threads)
add_test(NAME switch_drive_download_tests COMMAND switch_drive_download_tests)

option(SWITCHDRIVE_UI_PREVIEW "Build the host 1280x720 UI model preview" OFF)
if(SWITCHDRIVE_UI_PREVIEW)
    add_executable(switch_drive_ui_preview ui_preview.cpp ../switch/source/core.cpp ../switch/source/i18n.cpp ../switch/source/ui_model.cpp)
    target_include_directories(switch_drive_ui_preview PRIVATE ../switch/include ${JANSSON_INCLUDE_DIR})
    target_link_libraries(switch_drive_ui_preview PRIVATE ${JANSSON_LIBRARY})
endif()
