cmake_minimum_required(VERSION 3.24)
project(switch_drive_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(switch_drive_tests core_tests.cpp ../switch/source/core.cpp ../switch/source/installer.cpp ../switch/source/download.cpp ../switch/source/i18n.cpp ../switch/source/ui_model.cpp ../switch/source/qr.cpp)
target_include_directories(switch_drive_tests PRIVATE ../switch/include)
enable_testing()
add_test(NAME switch_drive_tests COMMAND switch_drive_tests)

option(SWITCHDRIVE_UI_PREVIEW "Build the host SDL UI renderer for visual review" OFF)
option(SWITCHDRIVE_UI_RUNTIME_TESTS "Test the Switch renderer with simulated display and input services" OFF)
if(SWITCHDRIVE_UI_PREVIEW OR SWITCHDRIVE_UI_RUNTIME_TESTS)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(SDL_UI REQUIRED IMPORTED_TARGET sdl2 SDL2_ttf)
endif()
if(SWITCHDRIVE_UI_PREVIEW)
    add_executable(switch_drive_ui_preview ui_preview.cpp ../switch/source/ui.cpp ../switch/source/ui_model.cpp ../switch/source/i18n.cpp ../switch/source/qr.cpp)
    target_include_directories(switch_drive_ui_preview PRIVATE ../switch/include)
    target_link_libraries(switch_drive_ui_preview PRIVATE PkgConfig::SDL_UI)
endif()

if(SWITCHDRIVE_UI_RUNTIME_TESTS)
    add_executable(switch_drive_ui_runtime_tests ui_runtime_tests.cpp ../switch/source/ui.cpp ../switch/source/ui_model.cpp ../switch/source/i18n.cpp ../switch/source/qr.cpp)
    target_compile_definitions(switch_drive_ui_runtime_tests PRIVATE __SWITCH__)
    target_include_directories(switch_drive_ui_runtime_tests PRIVATE switch_stubs ../switch/include)
    target_link_libraries(switch_drive_ui_runtime_tests PRIVATE PkgConfig::SDL_UI)
    add_test(NAME switch_drive_ui_runtime_tests COMMAND switch_drive_ui_runtime_tests)
endif()
